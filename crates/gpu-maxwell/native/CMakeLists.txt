cmake_minimum_required(VERSION 3.16)
project(nixe_mme LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_INSTALL_LIBDIR lib CACHE STRING "" FORCE)
set(ASMJIT_STATIC ON CACHE BOOL "" FORCE)
set(ASMJIT_TEST OFF CACHE BOOL "" FORCE)
set(ASMJIT_NO_AARCH64 ON CACHE BOOL "" FORCE)
set(ASMJIT_NO_COMPILER ON CACHE BOOL "" FORCE)
set(ASMJIT_NO_BUILDER ON CACHE BOOL "" FORCE)
set(ASMJIT_NO_UJIT ON CACHE BOOL "" FORCE)
set(ASMJIT_NO_CUSTOM_FLAGS ON CACHE BOOL "" FORCE)
add_subdirectory(../../../vendor/asmjit asmjit)
add_library(nixe_mme STATIC mme.cpp)
target_link_libraries(nixe_mme PRIVATE asmjit::asmjit)
if(NOT MSVC)
  target_compile_options(nixe_mme PRIVATE -fwrapv)
endif()
install(TARGETS nixe_mme ARCHIVE DESTINATION lib)
